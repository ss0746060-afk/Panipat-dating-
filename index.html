// अपनी index.html के पुराने processUpload फंक्शन की जगह सिर्फ इसे पेस्ट करें: function processUpload(event, mode) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(e) { const mediaUrl = e.target.result; const myName = localStorage.getItem('vip_name') || "अमित राव"; const myPic = localStorage.getItem('vip_pic') || "https://images.unsplash.com/photo-1535713875002-d1d0cf377fde?w=100"; const currentUser = localStorage.getItem('current_logged_in_user') || "राव_साहब_HR06"; let mediaHtml = ''; let isVideo = file.type.startsWith('video'); // 🟢 मुख्य बदलाव: अब हर पोस्ट के साथ 'owner' (मालिक का यूजरनेम) भी सेव होगा! let savedItems = JSON.parse(localStorage.getItem('my_uploaded_media') || '[]'); savedItems.unshift({ url: mediaUrl, type: mode === 'reel' ? 'reel' : (isVideo ? 'video' : 'photo'), owner: currentUser // यह ट्रैक करेगा कि कन्टेंट किसका है }); localStorage.setItem('my_uploaded_media', JSON.stringify(savedItems)); if (mode === 'reel') { mediaHtml = ``; } else if(isVideo) { mediaHtml = ``; } else { mediaHtml = ``; } const postHtml = `
${myName} ${mode === 'reel' ? '• रिल 🎬' : ''}
पानीपत • अभी-अभी
${mediaHtml}
1 लाइक
`; document.getElementById('mainFeed').insertAdjacentHTML('afterbegin', postHtml); alert(`${mode === 'reel' ? 'रील' : 'पोस्ट'} सफलतापूर्वक आपके पर्सनल अकाउंट से लाइव हो गई! 🚀`); }; reader.readAsDataURL(file); }
